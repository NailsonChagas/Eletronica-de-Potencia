Buck CCM
 
% Parametros de projeto
Vs = 200;
Vo = 150;
fs = 50 * 1e3;
P = 100;
delta_Vo = 0.1;
delta_Il = 0.3;
T = 1/fs;
 
% Calculo do duty cycle
D = Vo / Vs
D = 0.7500
 
% Corrente no indutor (iL == iR)
R = (Vo^2)/P
R = 225
iL = Vo/R
iL = 0.6667
 
% Corrente max e min no indutor
iL_min = iL - ((delta_Il * iL)/2)
iL_min = 0.5667
iL_max = iL + ((delta_Il * iL)/2)
iL_max = 0.7667
 
% Indutor
L = ((Vs - Vo) / (delta_Il * iL * fs) ) * D
L = 0.0037
 
% Indutancia mininima para CCM
Lmin = ((1 - D) * R * T) / 2
Lmin = 5.6250e-04
 
if L > Lmin
disp("L > Lmin: opera em CCM")
else
disp("L < Lmin: nao opera em CCM")
end

% Capacitor
C = (Vo * (1 - D)) / (8 * L * delta_Vo * Vo * fs^2)
